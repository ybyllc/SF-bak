1. 中值滤波（去脉冲噪声）
取连续5次测量，排序后取中值，有效剔除偶发跳变。

#define SAMPLE_NUM 5

float median_filter(float *samples, int n) {
    // 冒泡排序
    for(int i=0; i<n-1; i++)
        for(int j=0; j<n-i-1; j++)
            if(samples[j] > samples[j+1]) {
                float temp = samples[j];
                samples[j] = samples[j+1];
                samples[j+1] = temp;
            }
    return samples[n/2]; // 返回中值
}

#define WINDOW_SIZE 5
float moving_average(float new_val) {
    static float buffer[WINDOW_SIZE] = {0};
    static int idx = 0;
    float sum = 0;
    
    buffer[idx++] = new_val;
    if(idx >= WINDOW_SIZE) idx = 0;
    
    for(int i=0; i<WINDOW_SIZE; i++) sum += buffer[i];
    return sum / WINDOW_SIZE;
}

卡尔曼滤波（最优估计）
适合目标移动场景，平衡噪声抑制与响应速度。实测可提升7%精度。
推荐参数（经实验验证）：
测量噪声协方差 R = 100
过程噪声协方差 Q = 0.01

复制
typedef struct {
    float x;      // 估计值
    float p;      // 估计误差协方差
    float q;      // 过程噪声 (0.01)
    float r;      // 测量噪声 (100)
    float k;      // 卡尔曼增益
} KalmanFilter;

float kalman_update(KalmanFilter *kf, float measurement) {
    // 预测
    kf->p = kf->p + kf->q;
    // 更新
    kf->k = kf->p / (kf->p + kf->r);
    kf->x = kf->x + kf->k * (measurement - kf->x);
    kf->p = (1 - kf->k) * kf->p;
    return kf->x;
}
关键：Q和R差距过大（如Q=0.001, R=1000）会消除原始数据；差距过小则滤波效果不明显。

typedef struct {
    KalmanFilter kf;
    float last_valid;
    uint8_t sample_count;
} UltraSonicFilter;

float process_distance(UltraSonicFilter *f, float raw, float temp) {
    // 1. 温度补偿
    float compensated = raw * (331.3 + 0.606*temp) / 340.0;
    
    // 2. 异常值剔除
    if(!outlier_rejection(compensated, f->last_valid)) 
        return f->last_valid; // 保持上次值
    
    // 3. 中值滤波（5点）
    static float buf[5];
    buf[f->sample_count++ % 5] = compensated;
    if(f->sample_count < 5) return compensated;
    float median = median_filter(buf, 5);
    
    // 4. 卡尔曼滤波
    float filtered = kalman_update(&f->kf, median);
    f->last_valid = filtered;
    
    return filtered;
}
